14: Integer parse_unsigned(std::string_view text, std::string_view option, bool allow_zero = false) {
15:     Integer value = 0;
16:     const auto [end, error] = std::from_chars(text.data(), text.data() + text.size(), value);
17:     if (error != std::errc{} || end != text.data() + text.size() || (!allow_zero && value == 0)) {
18:         throw std::invalid_argument("Invalid value for " + std::string(option) +
19:                                     ": expected a representable " +
20:                                     (allow_zero ? "unsigned" : "positive") + " integer.");
21:     }
22:     return value;
23: }

25: double parse_real(std::string_view text, std::string_view option) {
26:     double value = 0;
27:     const auto [end, error] = std::from_chars(text.data(), text.data() + text.size(), value);
28:     if (error != std::errc{} || end != text.data() + text.size() || !std::isfinite(value))
29:         throw std::invalid_argument("Invalid value for " + std::string(option) +
30:                                     ": expected a finite number.");
31:     return value;

34: CliOptions parse_convert(std::span<const std::string_view> arguments) {
35:     CliOptions options;
36:     options.convert = true;
37:     bool have_dataset = false, have_out = false, have_documents = false;
38:     for (std::size_t i = 1; i < arguments.size(); ++i) {
39:         const auto name = arguments[i];
40:         if (name == "--help") {
41:             if (options.show_help) throw std::invalid_argument("Repeated option: --help.");
42:             options.show_help = true;
43:             continue;
44:         }
45:         bool* seen = name == "--dataset"         ? &have_dataset
46:                      : name == "--out"           ? &have_out
47:                      : name == "--documents-out" ? &have_documents
48:                                                  : nullptr;
49:         if (!seen) throw std::invalid_argument("Unknown option: " + std::string(name) + ".");
50:         if (*seen) throw std::invalid_argument("Repeated option: " + std::string(name) + ".");
51:         *seen = true;
52:         if (i + 1 == arguments.size() || arguments[i + 1].empty() ||
53:             arguments[i + 1].starts_with("--"))
54:             throw std::invalid_argument("Missing value for " + std::string(name) + ".");
55:         const std::string value(arguments[++i]);
56:         if (name == "--dataset")
57:             options.conversion.dataset_path = value;
58:         else if (name == "--out")
59:             options.conversion.output_path = value;
60:         else
61:             options.conversion.documents_output = value;
62:     }
63:     if (!options.show_help && (!have_dataset || !have_out))
64:         throw std::invalid_argument("convert requires --dataset FILE and --out FILE.avec.");
65:     if (!options.show_help) validate_convert_request(options.conversion);
66:     return options;
67: }

72:     CliOptions options;
73:     if (arguments.empty() || (arguments.size() == 1 && arguments[0] == "--help")) {

107:     bool have_bm25_strategy = false;
108:     bool have_analyzer = false;
109:     bool have_query_weighting = false;
110:     bool bm25_selected = false;
111:     for (std::size_t i = 1; i < arguments.size(); ++i) {
112:         const auto name = arguments[i];
113:         if (name == "--help") {
114:             if (options.show_help) {
115:                 throw std::invalid_argument("Repeated option: --help.");
116:             }
117:             options.show_help = true;
118:             continue;
119:         }
120:         // Valueless switches.
121:         if (name == "--all-queries" || name == "--parallel-retrievers" || name == "--drop-fp32") {
122:             bool& flag = name == "--all-queries"           ? options.search.all_queries
123:                          : name == "--parallel-retrievers" ? options.search.parallel_retrievers

125:             if (flag) throw std::invalid_argument("Repeated option: " + std::string(name) + ".");
126:             flag = true;
127:             continue;

129:         // Repeatable filters (combined with AND).
130:         if (name == "--filter" || name == "--filter-date") {
131:             if (i + 1 == arguments.size() || arguments[i + 1].starts_with("--"))
132:                 throw std::invalid_argument("Missing value for " + std::string(name) + ".");
133:             const auto value = arguments[++i];

135:                                                                 : parse_date_filter(value));

139:         if (name == "--dataset") {
140:             seen = &have_dataset;
141:         } else if (name == "--query") {
142:             seen = &have_query;
143:         } else if (name == "--mode") {
144:             seen = &have_mode;
145:         } else if (name == "--query-text") {
146:             seen = &have_query_text;
147:         } else if (name == "--query-id") {
148:             seen = &have_query_id;
149:         } else if (name == "--bm25-k1") {
150:             seen = &have_k1;
151:         } else if (name == "--bm25-b") {
152:             seen = &have_b;
153:         } else if (name == "--candidate-k") {
154:             seen = &have_candidate_k;
155:         } else if (name == "--rrf-k") {
156:             seen = &have_rrf_k;
157:         } else if (name == "--rrf-weight") {
158:             seen = &have_rrf_weight;
159:         } else if (name == "--alpha") {
160:             seen = &have_alpha;
161:         } else if (name == "--vectors") {
162:             seen = &have_vectors;
163:         } else if (name == "--build-threads") {
164:             seen = &have_build_threads;
165:         } else if (name == "--query-threads") {
166:             seen = &have_query_threads;
167:         } else if (name == "--quantization") {
168:             seen = &have_quantization;
169:         } else if (name == "--rerank") {
170:             seen = &have_rerank;
171:         } else if (name == "--pq-subspaces") {
172:             seen = &have_pq_subspaces;
173:         } else if (name == "--bm25-strategy") {
174:             seen = &have_bm25_strategy;
175:         } else if (name == "--analyzer") {
176:             seen = &have_analyzer;
177:         } else if (name == "--bm25-query-weighting") {
178:             seen = &have_query_weighting;
179:         } else if (name == "--algorithm") {
180:             seen = &have_algorithm;
181:         } else if (name == "--k") {
182:             seen = &have_k;
183:         } else if (name == "--metric") {
184:             seen = &have_metric;
185:         } else if (name == "--m") {
186:             seen = &have_m;
187:         } else if (name == "--m0") {
188:             seen = &have_m0;
189:         } else if (name == "--ef-construction") {
190:             seen = &have_ef_construction;
191:         } else if (name == "--ef-search") {
192:             seen = &have_ef_search;
193:         } else if (name == "--seed") {
194:             seen = &have_seed;
195:         } else if (name == "--neighbor-selection") {
196:             seen = &have_selection;
197:         } else {
198:             throw std::invalid_argument("Unknown option: " + std::string(name) + ".");
199:         }
200:         if (*seen) {
201:             throw std::invalid_argument("Repeated option: " + std::string(name) + ".");
202:         }
203:         *seen = true;
204:         if (i + 1 == arguments.size() || (arguments[i + 1].empty() && name != "--query-text") ||
205:             arguments[i + 1].starts_with("--")) {
206:             throw std::invalid_argument("Missing value for " + std::string(name) + ".");
207:         }
208:         const auto value = arguments[++i];
209:         if (name == "--dataset") {
210:             options.search.dataset_path = std::string(value);
211:         } else if (name == "--query") {
212:             options.search.query_path = std::string(value);
213:         } else if (name == "--query-text") {
214:             options.search.query_text = std::string(value);
215:         } else if (name == "--query-id") {
216:             options.search.query_id = std::string(value);
217:         } else if (name == "--mode") {
218:             if (value == "vector")
219:                 options.search.mode = SearchMode::Vector;
220:             else if (value == "lexical")
221:                 options.search.mode = SearchMode::Lexical;
222:             else if (value == "hybrid")
223:                 options.search.mode = SearchMode::Hybrid;
224:             else
225:                 throw std::invalid_argument("Invalid --mode: use vector, lexical or hybrid.");
226:         } else if (name == "--bm25-k1") {
227:             options.search.bm25_config.k1 = parse_real(value, name);
228:         } else if (name == "--bm25-b") {
229:             options.search.bm25_config.b = parse_real(value, name);
230:         } else if (name == "--rrf-k") {
231:             options.search.rrf_config.k_rrf = parse_real(value, name);
232:         } else if (name == "--rrf-weight") {
233:             options.search.rrf_config.dense_weight = parse_real(value, name);
234:         } else if (name == "--alpha") {
235:             options.search.rrf_config.method = hybrid::FusionMethod::Convex;
236:             options.search.rrf_config.alpha = parse_real(value, name);
237:         } else if (name == "--vectors") {
238:             options.search.vectors_path = std::string(value);
239:         } else if (name == "--build-threads") {
240:             options.search.build_threads = parse_unsigned<std::size_t>(value, name);
241:         } else if (name == "--query-threads") {
242:             options.search.query_threads = parse_unsigned<std::size_t>(value, name);
243:         } else if (name == "--quantization") {
244:             if (value == "none")
245:                 options.search.quantization = Quantization::None;
246:             else if (value == "sq8")
247:                 options.search.quantization = Quantization::Sq8;
248:             else if (value == "pq")
249:                 options.search.quantization = Quantization::Pq;
250:             else
251:                 throw std::invalid_argument("Invalid --quantization: use none, sq8 or pq.");
252:         } else if (name == "--rerank") {
253:             options.search.rerank = parse_unsigned<std::size_t>(value, name, true);
254:         } else if (name == "--pq-subspaces") {
255:             options.search.pq_subspaces = parse_unsigned<std::size_t>(value, name);
256:         } else if (name == "--bm25-strategy") {
257:             auto& strategy = options.search.bm25_config.strategy;
258:             if (value == "exhaustive")
259:                 strategy = lexical::Bm25Strategy::Exhaustive;
260:             else if (value == "taat")
261:                 strategy = lexical::Bm25Strategy::TermAtATime;
262:             else if (value == "wand")
263:                 strategy = lexical::Bm25Strategy::Wand;
264:             else if (value == "bmw")
265:                 strategy = lexical::Bm25Strategy::BlockMaxWand;
266:             else if (value == "maxscore")
267:                 strategy = lexical::Bm25Strategy::MaxScore;
268:             else
269:                 throw std::invalid_argument(
270:                     "Invalid --bm25-strategy: use exhaustive, taat, wand, bmw or maxscore.");
271:         } else if (name == "--analyzer") {
272:             if (value == "ascii-tech-v1")
273:                 options.search.bm25_config.analyzer = lexical::Analyzer::AsciiTechV1;
274:             else if (value == "english-v1")
275:                 options.search.bm25_config.analyzer = lexical::Analyzer::EnglishV1;
276:             else
277:                 throw std::invalid_argument("Invalid --analyzer: use ascii-tech-v1 or english-v1.");
278:         } else if (name == "--bm25-query-weighting") {
279:             if (value == "unique")
280:                 options.search.bm25_config.query_weighting = lexical::QueryTermWeighting::Unique;
281:             else if (value == "frequency")
282:                 options.search.bm25_config.query_weighting = lexical::QueryTermWeighting::Frequency;
283:             else
284:                 throw std::invalid_argument(
285:                     "Invalid --bm25-query-weighting: use unique or frequency.");
286:         } else if (name == "--candidate-k") {
287:             options.search.candidate_k = parse_unsigned<std::size_t>(value, name);
288:         } else if (name == "--algorithm") {
289:             if (value == "brute-force") {
290:                 options.search.algorithm = SearchAlgorithm::BruteForce;
291:             } else if (value == "hnsw") {
292:                 options.search.algorithm = SearchAlgorithm::Hnsw;
293:             } else if (value == "bm25") {
294:                 bm25_selected = true;
295:             } else {
296:                 throw std::invalid_argument("Unknown algorithm: " + std::string(value) +
297:                                             ". Available algorithms: brute-force, hnsw, bm25.");
298:             }
299:         } else if (name == "--metric") {
300:             if (value == "cosine") {
301:                 options.search.metric = DistanceKind::Cosine;
302:             } else if (value == "squared-euclidean") {
303:                 options.search.metric = DistanceKind::SquaredEuclidean;
304:             } else {
305:                 throw std::invalid_argument("Invalid --metric: use cosine or squared-euclidean.");
306:             }
307:         } else if (name == "--neighbor-selection") {
308:             if (value == "simple")
309:                 options.search.hnsw_config.neighbor_selection = NeighborSelection::Simple;
310:             else if (value == "heuristic")
311:                 options.search.hnsw_config.neighbor_selection = NeighborSelection::Heuristic;
312:             else
313:                 throw std::invalid_argument(
314:                     "Invalid --neighbor-selection: use simple or heuristic.");
315:         } else if (name == "--k") {
316:             options.search.k = parse_unsigned<std::size_t>(value, name);
317:         } else if (name == "--seed") {
318:             options.search.hnsw_config.random_seed =
319:                 parse_unsigned<std::uint64_t>(value, name, true);
320:         } else {
321:             const auto parameter = parse_unsigned<std::size_t>(value, name);
322:             auto& config = options.search.hnsw_config;
323:             if (name == "--m")
324:                 config.m = parameter;
325:             else if (name == "--m0")
326:                 config.m_max_0 = parameter;
327:             else if (name == "--ef-construction")
328:                 config.ef_construction = parameter;
329:             else if (name == "--ef-search")
330:                 config.ef_search = parameter;
331:         }
332:     }
333:     if (!options.show_help) {
334:         if (!have_dataset) {
335:             throw std::invalid_argument("Missing required option: --dataset.");
336:         }
337:         if (!have_query && !have_query_text) {
338:             throw std::invalid_argument("Missing required option: --query.");
339:         }
340:         const auto mode = options.search.mode;
341:         if (!have_mode && bm25_selected) options.search.mode = SearchMode::Lexical;
342:         if (options.search.mode == SearchMode::Vector && !have_algorithm) {
343:             throw std::invalid_argument("Missing required option: --algorithm.");
344:         }
345:         if ((options.search.mode == SearchMode::Lexical && have_algorithm && !bm25_selected) ||
346:             (options.search.mode == SearchMode::Vector && bm25_selected))
347:             throw std::invalid_argument(
348:                 "Incompatible --mode and --algorithm: BM25 is lexical; brute-force and hnsw are "
349:                 "vector algorithms.");
350:         // Hybrid always fuses HNSW with BM25; --algorithm hnsw is accepted as a restatement.
351:         if (mode == SearchMode::Hybrid) {
352:             if (have_algorithm &&
353:                 (bm25_selected || options.search.algorithm != SearchAlgorithm::Hnsw))
354:                 throw std::invalid_argument(
355:                     "Incompatible --mode and --algorithm: hybrid fuses HNSW with BM25.");
356:             options.search.algorithm = SearchAlgorithm::Hnsw;
357:         }
358:         if (options.search.mode == SearchMode::Lexical && have_metric)
359:             throw std::invalid_argument(
360:                 "--metric requires vector or hybrid mode; BM25 returns scores, not distances.");
361:         if (options.search.mode == SearchMode::Vector &&
362:             (have_k1 || have_b || have_bm25_strategy || have_analyzer || have_query_weighting))
363:             throw std::invalid_argument(
364:                 "BM25 options require lexical or hybrid mode (--mode lexical|hybrid or "
365:                 "--algorithm bm25).");
366:         if (options.search.mode != SearchMode::Hybrid &&
367:             (have_candidate_k || have_rrf_k || have_rrf_weight || have_alpha || have_vectors))
368:             throw std::invalid_argument(
369:                 "--candidate-k, --rrf-k, --rrf-weight, --alpha and --vectors require --mode "
370:                 "hybrid.");
371:         if (have_alpha && (have_rrf_k || have_rrf_weight))
372:             throw std::invalid_argument(
373:                 "--alpha selects convex fusion; it cannot be combined with --rrf-k or "
374:                 "--rrf-weight.");
375:         const bool hnsw_selected = options.search.mode == SearchMode::Hybrid ||
376:                                    (options.search.mode == SearchMode::Vector &&
377:                                     options.search.algorithm == SearchAlgorithm::Hnsw);
378:         if (!hnsw_selected && (have_m || have_m0 || have_ef_construction || have_ef_search ||
379:                                have_seed || have_selection || have_build_threads)) {
380:             throw std::invalid_argument("HNSW options require --algorithm hnsw or --mode hybrid.");
381:         }
382:         if (have_query_threads && !options.search.all_queries)
383:             throw std::invalid_argument("--query-threads requires --all-queries.");
384:         validate_search_request(options.search);
385:     }
386:     return options;
387: }

48: void print_table(std::ostream& output, const Row& headings, const std::vector<Row>& rows) {
49:     std::vector<std::size_t> widths;
50:     for (const auto& heading : headings) widths.push_back(heading.size());
51:     for (const auto& row : rows)
52:         for (std::size_t column = 0; column < widths.size(); ++column)
53:             widths[column] = std::max(widths[column], row[column].size());
54:     const auto border = [&] {
55:         output << '+';
56:         for (auto width : widths) output << std::string(width + 2, '-') << '+';
57:         output << '\n';
58:     };
59:     const auto print_row = [&](const Row& row, bool right) {
60:         output << '|';
61:         for (std::size_t column = 0; column < widths.size(); ++column) {
62:             const std::string padding(widths[column] - row[column].size(), ' ');
63:             output << ' ' << (right ? padding + row[column] : row[column] + padding) << " |";
64:         }
65:         output << '\n';
66:     };
67:     border();
68:     print_row(headings, false);
69:     border();
70:     for (const auto& row : rows) {
71:         print_row(row, true);
72:         border();
73:     }
74: }

271: int run(std::span<const std::string_view> arguments, std::ostream& output, std::ostream& errors) {
272:     CliOptions options;
273:     try {
274:         options = parse_arguments(arguments);
275:     } catch (const std::invalid_argument& error) {
276:         errors << "Error: " << error.what() << "\nUse aetherindex --help for usage.\n";
277:         return 2;
278:     } catch (const std::exception& error) {
279:         errors << "Error: " << error.what() << '\n';
280:         return 4;
281:     }
282:     try {
283:         if (options.show_help) {
284:             output << help_text();
285:             return 0;
286:         }
287:         if (options.convert) {
288:             const auto converted = convert_vector_file(options.conversion);
289:             std::ostringstream formatted;
290:             formatted.imbue(std::locale::classic());
291:             formatted << "AetherIndex Convert\n\nVectors written: " << converted.vectors
292:                       << "\nDimensions: " << converted.dimension
293:                       << "\nFormat: .avec v1 (raw FP32, little-endian, CRC-32C)";
294:             if (converted.documents_written)
295:                 formatted << "\nDocuments written (text only): " << converted.documents_written;
296:             formatted << "\n\nConversion completed successfully.\n";
297:             output << formatted.str();
298:             return 0;
299:         }
300:         const auto response = SearchService{}.search(options.search);
301:         std::ostringstream formatted;
302:         formatted.imbue(std::locale::classic());
303:         formatted << "AetherIndex Search\n\n";
304:         const auto print = [&](const SearchResponse& result) {
305:             if (options.search.mode == SearchMode::Lexical)
306:                 print_lexical(formatted, options.search, result);
307:             else if (options.search.mode == SearchMode::Hybrid)
308:                 print_hybrid(formatted, options.search, result);
309:             else
310:                 print_vector(formatted, options.search, result);
311:         };
312:         if (options.search.all_queries) {
313:             const double seconds = response.batch_time.count() / 1000.0;
314:             formatted << "Batch queries: " << response.batch.size()
315:                       << "\nQuery threads: " << options.search.query_threads
316:                       << "\nBatch time (ms): " << number(response.batch_time.count(), 6)
317:                       << "\nThroughput (queries/s): "
318:                       << number(seconds > 0 ? response.batch.size() / seconds : 0, 1)
319:                       << "\nBatch time scope: all queries, excluding input loading, index "
320:                          "builds and output\n";
321:             // Each block is exactly the single-query report for that query.
322:             for (std::size_t i = 0; i < response.batch.size(); ++i) {
323:                 auto view = response.batch[i];
324:                 view.vectors_indexed = response.vectors_indexed;
325:                 view.documents_indexed = response.documents_indexed;
326:                 view.dimension = response.dimension;
327:                 view.index_build_time = response.index_build_time;
328:                 view.lexical_build_time = response.lexical_build_time;
329:                 view.quantization_time = response.quantization_time;
330:                 view.fp32_bytes = response.fp32_bytes;
331:                 view.code_bytes = response.code_bytes;
332:                 view.fp32_released = response.fp32_released;
333:                 view.rerank = response.rerank;
334:                 view.filtered = response.filtered;
335:                 view.filter_matches = response.filter_matches;
336:                 view.filter_population = response.filter_population;
337:                 view.exact_filter = response.exact_filter;
338:                 formatted << "\n=== Query " << i + 1 << "/" << response.batch.size() << " ===\n";
339:                 print(view);
340:             }
341:         } else {
342:             print(response);
343:         }
344:         formatted << "\nSearch completed successfully.\n";
345:         output << formatted.str();
346:         return 0;
347:     } catch (const InputDataError& error) {
348:         errors << "Error: " << error.what() << '\n';
349:         return 3;
350:     } catch (const std::exception& error) {
351:         errors << "Error during search: " << error.what() << '\n';
352:         return 4;
353:     }
